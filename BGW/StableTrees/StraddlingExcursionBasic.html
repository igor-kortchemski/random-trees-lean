                    theorem BGW.StableTrees.ae_normalize_straddling_path {θ : ℝ} (hθ : 1 < θ ∧ θ ≤ 2) {t : NNReal} (ht : 0 < t) :
                    ∀ᵐ (f : Skorokhod.DLoc) ∂↑(stableLocalLaw θ), ∀ (s : NNReal), ↑(normalizeExc θ (straddlingExcursion t f)).path s = straddleLife t f ^ (-(1 / θ)) * (↑f (min (straddleStart t f + (straddleEnd t f - straddleStart t f) * s) (straddleEnd t f)) - ↑f (straddleStart t f))
