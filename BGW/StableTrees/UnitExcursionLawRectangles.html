    theorem BGW.StableTrees.excursionMeasure_shape_tail_rectangle {θ : ℝ} (hθ : 1 < θ ∧ θ ≤ 2) {A : Set Excursion} (hA : MeasurableSet A) (t : ℝ) (ht : 0 < t) :
    (excursionMeasure θ) ({e : Excursion | normalizeExc θ e ∈ A} ∩ {e : Excursion | t < ↑e.lifetime}) = ENNReal.ofReal (t ^ (-(1 / θ)) / Real.Gamma (1 - 1 / θ)) * ↑(unitExcursionLaw θ) A
